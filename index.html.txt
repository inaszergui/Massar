                <div className="absolute inset-x-0 top-0 bottom-0 pointer-events-none bg-repeating-linear bg-neutral-100/30" style={{ backgroundImage: 'linear-gradient(#f9fafb 1px, transparent 1px)', backgroundSize: '100% 2rem', marginTop: '1.5rem' }}></div>
                
                <textarea
                  value={quickNote}
                  onChange={(e) => setQuickNote(e.target.value)}
                  placeholder="ابدئي بالكتابة الحرّة هنا..."
                  className="w-full min-h-[400px] text-neutral-700 bg-transparent relative z-10 focus:outline-none text-base leading-[2rem] font-serif resize-y"
                  style={{ lineHeight: '2rem' }}
                />
              </div>

              <div className="bg-neutral-50 px-6 py-3 border-t border-rose-100/40 flex flex-col sm:flex-row items-center justify-between gap-2">
                <span className="text-xs text-neutral-400">
                  عدد الحروف المكتوبة: <strong className="text-rose-500">{quickNote.length}</strong> حرف
                </span>
                <span className="text-xs text-neutral-400 flex items-center gap-1">
                  <span className="w-2 h-2 rounded-full bg-emerald-400 inline-block"></span>
                  نشط وجاهز للتعديل الفوري
                </span>
              </div>
            </div>
          </div>
        )}

        {/* ----------------- 5. DAILY TASKS TAB ----------------- */}
        {activeTab === 'tasks' && (
          <div className="space-y-6 animate-fade-in">
            <div className="flex flex-col md:flex-row md:items-center justify-between gap-2 border-b border-rose-100 pb-4">
              <div>
                <h2 className="text-xl font-bold text-neutral-800 flex items-center gap-2">
                  <CheckCircle className="w-5 h-5 text-rose-500" />
                  تتبع المهام اليومية والخطوات
                </h2>
                <p className="text-xs text-neutral-400">النجاح هو تراكم خطواتكِ البسيطة المتقنة كل يوم</p>
              </div>
              <span className="text-xs bg-rose-50 text-rose-600 px-3 py-1 rounded-full self-start md:self-auto">
                اكتمل اليوم: {completedCount} من أصل {tasks.length}
              </span>
            </div>

            <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
              
              <div className="bg-white p-6 rounded-3xl border border-rose-100/40 shadow-sm flex flex-col justify-between gap-6 h-fit">
                <div>
                  <h3 className="font-bold text-neutral-800 mb-1 flex items-center gap-2">
                    📊
                    مستوى الإنجاز لليوم
                  </h3>
                  <p className="text-xs text-neutral-400 mb-6">مؤشر التقدم الإيجابي لخطواتكِ الشخصية</p>
                  
                  <div className="space-y-4">
                    <div className="relative pt-1">
                      <div className="flex mb-2 items-center justify-between">
                        <div>
                          <span className="text-xs font-semibold inline-block py-1 px-2.5 uppercase rounded-full text-rose-600 bg-rose-50">
                            معدل الإنجاز الحالي
                          </span>
                        </div>
                        <div className="text-right">
                          <span className="text-sm font-bold inline-block text-rose-600">
                            {progressPercent}%
                          </span>
                        </div>
                      </div>
                      <div className="overflow-hidden h-3 mb-4 text-xs flex rounded-full bg-rose-100/50">
                        <div 
                          style={{ width: `${progressPercent}%` }} 
                          className="shadow-none flex flex-col text-center whitespace-nowrap text-white justify-center bg-gradient-to-r from-rose-400 to-pink-500 transition-all duration-500"
                        ></div>
                      </div>
                    </div>
                  </div>
                </div>

                <div className="p-4 rounded-2xl bg-gradient-to-tr from-rose-50/50 to-pink-50/50 border border-rose-100/50">
                  <h4 className="text-xs font-bold text-rose-500 mb-1">💡 كلمة لكِ:</h4>
                  <p className="text-xs text-neutral-600 leading-relaxed">
                    {progressPercent === 100 
                      ? "رائعة ومذهلة! لقد أنجزتِ كل مهامكِ لليوم بنجاح، استرخي واستمتعي بوقتكِ الآن."
                      : progressPercent >= 50 
                      ? "إنجاز رائع! لقد قطعتِ منتصف الطريق، خطوات بسيطة تفصلكِ عن الاكتمال التام."
                      : progressPercent > 0 
                      ? "بداية مُمتازة لليوم، تذكري دائماً أن خطوة واحدة بسيطة أفضل من السكون الكلي."
                      : "يومكِ يبدأ الآن بلطف، حددي مهامكِ البسيطة وابدئي الإنجاز الهادئ."}
                  </p>
                </div>
              </div>

              <div className="lg:col-span-2 space-y-4">
                
                <form onSubmit={handleAddTask} className="flex gap-2 bg-white p-2 rounded-2xl border border-rose-100/40 shadow-sm">
                  <input 
                    type="text" 
                    placeholder="اكتبي مهمة يومية جديدة هنا برفق..."
                    value={newTaskText}
                    onChange={(e) => setNewTaskText(e.target.value)}
                    className="flex-grow px-4 py-2 bg-transparent text-sm focus:outline-none"
                  />
                  <button 
                    type="submit"
                    className="bg-rose-500 hover:bg-rose-600 text-white px-5 py-2.5 rounded-xl font-medium transition-colors text-xs flex items-center gap-1.5 shadow-sm shadow-rose-200"
                  >
                    <Plus className="w-3.5 h-3.5" />
                    إضافة
                  </button>
                </form>

                <div className="bg-white rounded-3xl border border-rose-100/40 shadow-sm p-6 space-y-3">
                  {tasks.length === 0 ? (
                    <div className="text-center py-8 text-neutral-400">
                      <p className="text-sm">لا توجد مهام مضافة في مفكرتكِ بعد.</p>
                      <p className="text-xs text-neutral-300 mt-1">ابدئي كتابة قائمة أعمالكِ البسيطة بالأعلى.</p>
                    </div>
                  ) : (
                    tasks.map((task) => (
                      <div 
                        key={task.id} 
                        className={`flex items-center justify-between p-3.5 rounded-2xl border transition-all ${
                          task.completed 
                            ? 'bg-neutral-50/60 border-neutral-100 text-neutral-400 line-through' 
                            : 'bg-white border-rose-100/40 text-neutral-700 hover:border-rose-200'
                        }`}
                      >
                        <div 
                          onClick={() => toggleTask(task.id)}
                          className="flex items-center gap-3 cursor-pointer flex-grow"
                        >
                          <div className={`w-5 h-5 rounded-full border-2 flex items-center justify-center transition-all flex-shrink-0 ${
                            task.completed 
                              ? 'border-emerald-500 bg-emerald-500 text-white' 
                              : 'border-rose-300 bg-white hover:border-rose-500'
                          }`}>
                            {task.completed && <span className="text-[10px] font-bold">✓</span>}
                          </div>
                          <span className="text-sm font-medium leading-relaxed">{task.text}</span>
                        </div>

                        <button 
                          onClick={() => handleDeleteTask(task.id)}
                          className="p-1 text-neutral-300 hover:text-rose-500 hover:bg-rose-50 rounded-lg transition-colors flex-shrink-0"
                        >
                          <Trash2 className="w-4 h-4" />
                        </button>
                      </div>
                    ))
                  )}
                </div>

              </div>

            </div>
          </div>
        )}

        {/* ----------------- 6. LETTERS TO SELF TAB ----------------- */}
        {activeTab === 'letters' && (
          <div className="space-y-6 animate-fade-in">
            <div className="flex flex-col md:flex-row md:items-center justify-between gap-2 border-b border-rose-100 pb-4">
              <div>
                <h2 className="text-xl font-bold text-neutral-800 flex items-center gap-2">
                  <BookOpen className="w-5 h-5 text-rose-500" />
                  مساحة الرسائل الوجدانية العميقة
                </h2>
                <p className="text-xs text-neutral-400">ملاذكِ الشخصي للبوح بصدق والامتنان لذاتكِ في مختلف محطاتها العمرية</p>
              </div>
            </div>

            {editingLetter ? (
              <div className="bg-white rounded-3xl border border-rose-100/40 shadow-sm overflow-hidden animate-fade-in">
                <div className="bg-gradient-to-r from-rose-500 to-pink-500 px-6 py-4 text-white flex justify-between items-center">
                  <h3 className="font-bold text-sm sm:text-base">
                    {editingLetter === 'youngerSelf' ? '✍️ تعديل الرسالة للطفلة التي بداخلكِ' : '✍️ تعديل الرسالة لتعزيز قوتكِ الحالية'}
                  </h3>
                  <button 
                    onClick={() => setEditingLetter(null)}
                    className="text-xs bg-white/20 hover:bg-white/30 text-white px-3 py-1.5 rounded-xl font-medium transition-colors"
                  >
                    إلغاء
                  </button>
                </div>

                <div className="p-6">
                  <textarea
                    rows="12"
                    value={tempLetterText}
                    onChange={(e) => setTempLetterText(e.target.value)}
                    className="w-full text-neutral-700 bg-stone-50/50 border border-rose-100/50 rounded-2xl p-4 text-base leading-relaxed font-serif focus:outline-none focus:border-rose-400 focus:ring-1 focus:ring-rose-200 resize-none"
                    placeholder="اكتبي مشاعركِ وامتنانكِ بلطف وود هنا..."
                  ></textarea>

                  <div className="mt-4 flex justify-end gap-2">
                    <button 
                      onClick={() => setEditingLetter(null)}
                      className="px-4 py-2 rounded-xl text-xs font-semibold text-neutral-400 hover:text-neutral-600 bg-neutral-100 transition-colors"
                    >
                      تراجع
                    </button>
                    <button 
                      onClick={saveLetter}
                      className="px-5 py-2 rounded-xl text-xs font-semibold text-white bg-rose-500 hover:bg-rose-600 shadow-sm shadow-rose-200 transition-colors"
                    >
                      حفظ الرسالة الوجدانية
                    </button>
                  </div>
                </div>
              </div>
            ) : (
              <div className="grid grid-cols-1 md:grid-cols-2 gap-8">
                
                {/* Letter to Younger Self */}
                <div className="bg-white rounded-3xl border border-rose-100/40 shadow-sm overflow-hidden flex flex-col justify-between">
                  <div className="bg-rose-50 p-6 flex items-center justify-between border-b border-rose-100/40">
                    <div className="flex items-center gap-3">
                      <div className="w-10 h-10 rounded-full bg-rose-100 text-rose-600 flex items-center justify-center font-bold text-lg">
                        🧸
                      </div>
                      <div>
                        <h3 className="font-bold text-neutral-800 text-base">رسالة لنفسي وأنا صغيرة</h3>
                        <p className="text-[10px] text-neutral-400">آخر تعديل: {letters.youngerSelf.lastUpdated}</p>
                      </div>
                    </div>
                  </div>

                  <div className="p-6 flex-grow">
                    <div className="bg-stone-50/70 p-4 rounded-2xl border border-stone-100 font-serif text-sm text-neutral-600 leading-relaxed whitespace-pre-line min-h-[160px] max-h-[220px] overflow-y-auto font-normal">
                      {letters.youngerSelf.text}
                    </div>
                  </div>

                  <div className="bg-neutral-50 px-6 py-4 border-t border-rose-100/20 flex justify-between items-center">
                    <span className="text-[10px] text-rose-500/80 font-semibold">رسالة دافئة للطفلة التي بداخلكِ 🌸</span>
                    <button 
                      onClick={() => startEditingLetter('youngerSelf')}
                      className="text-xs bg-rose-50 hover:bg-rose-100 text-rose-600 px-4 py-2 rounded-xl font-bold transition-colors flex items-center gap-1.5"
                    >
                      <Edit3 className="w-3.5 h-3.5" />
                      تعديل وبوح
                    </button>
                  </div>
                </div>

                {/* Letter to Present Self */}
                <div className="bg-white rounded-3xl border border-rose-100/40 shadow-sm overflow-hidden flex flex-col justify-between">
                  <div className="bg-pink-50 p-6 flex items-center justify-between border-b border-pink-100/40">
                    <div className="flex items-center gap-3">
                      <div className="w-10 h-10 rounded-full bg-pink-100 text-pink-600 flex items-center justify-center font-bold text-lg">
                        👑
                      </div>
                      <div>
                        <h3 className="font-bold text-neutral-800 text-base">رسالة لنفسي في الحاضر</h3>
                        <p className="text-[10px] text-neutral-400">آخر تعديل: {letters.presentSelf.lastUpdated}</p>
                      </div>
                    </div>
                  </div>

                  <div className="p-6 flex-grow">
                    <div className="bg-stone-50/70 p-4 rounded-2xl border border-stone-100 font-serif text-sm text-neutral-600 leading-relaxed whitespace-pre-line min-h-[160px] max-h-[220px] overflow-y-auto">
                      {letters.presentSelf.text}
                    </div>
                  </div>

                  <div className="bg-neutral-50 px-6 py-4 border-t border-rose-100/20 flex justify-between items-center">
                    <span className="text-[10px] text-pink-500/80 font-semibold">لتعزيز قوتكِ وامتنانكِ الحالي ✨</span>
                    <button 
                      onClick={() => startEditingLetter('presentSelf')}
                      className="text-xs bg-pink-50 hover:bg-pink-100 text-pink-600 px-4 py-2 rounded-xl font-bold transition-colors flex items-center gap-1.5"
                    >
                      <Edit3 className="w-3.5 h-3.5" />
                      تعديل وبوح
                    </button>
                  </div>
                </div>

              </div>
            )}
          </div>
        )}

      </main>

      {/* FOOTER SECTION */}
      <footer className="bg-white border-t border-rose-100/40 py-6 text-center text-xs text-neutral-400 mt-12 pb-24 lg:pb-6">
        <div className="max-w-5xl mx-auto px-4 space-y-2">
          <p className="font-semibold text-rose-500">مسار | Masar © 2026</p>
          <p>جميع بياناتكِ الاستراتيجية والإبداعية والوجدانية محفوظة بأمان وخصوصية تامة على جهازكِ.</p>
          <div className="flex justify-center gap-3 text-rose-300 pt-2">
            <span>• رؤية واضحة</span>
            <span>• قيم ثابتة</span>
            <span>• وعي نقي</span>
          </div>
        </div>
      </footer>

      {/* Mobile Navigation Bar */}
      <div className="lg:hidden fixed bottom-0 inset-x-0 z-50 bg-white/95 backdrop-blur-md border-t border-rose-100/80 px-2 py-2 flex justify-around items-center shadow-lg shadow-rose-300/20">
        <button 
          onClick={() => setActiveTab('home')}
          className={`flex flex-col items-center gap-1 p-2 rounded-xl transition-all ${activeTab === 'home' ? 'text-rose-600 scale-105 font-bold' : 'text-neutral-400'}`}
        >
          <Heart className="w-5 h-5 fill-current" />
          <span className="text-[10px]">الرئيسية</span>
        </button>

        <button 
          onClick={() => setActiveTab('strategy')}
          className={`flex flex-col items-center gap-1 p-2 rounded-xl transition-all ${activeTab === 'strategy' ? 'text-rose-600 scale-105 font-bold' : 'text-neutral-400'}`}
        >
          <Compass className="w-5 h-5" />
          <span className="text-[10px]">التخطيط</span>
        </button>

        <button 
          onClick={() => setActiveTab('brainstorming')}
          className={`flex flex-col items-center gap-1 p-2 rounded-xl transition-all ${activeTab === 'brainstorming' ? 'text-rose-600 scale-105 font-bold' : 'text-neutral-400'}`}
        >
          <Sparkles className="w-5 h-5" />
          <span className="text-[10px]">الأفكار</span>
        </button>

        <button 
          onClick={() => setActiveTab('tasks')}
          className={`flex flex-col items-center gap-1 p-2 rounded-xl transition-all ${activeTab === 'tasks' ? 'text-rose-600 scale-105 font-bold' : 'text-neutral-400'}`}
        >
          <CheckCircle className="w-5 h-5" />
          <span className="text-[10px]">المهام</span>
        </button>

        <button 
          onClick={() => setActiveTab('letters')}
          className={`flex flex-col items-center gap-1 p-2 rounded-xl transition-all ${activeTab === 'letters' ? 'text-rose-600 scale-105 font-bold' : 'text-neutral-400'}`}
        >
          <BookOpen className="w-5 h-5" />
          <span className="text-[10px]">الرسائل</span>
        </button>
      </div>
    </div>
  );
}
